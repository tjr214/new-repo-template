from __future__ import annotations

from dataclasses import dataclass

from python_lib import build_greeting


@dataclass(frozen=True)
class StarterSummary:
    title: str
    greeting: str
    commands: tuple[str, ...]


def build_starter_summary(subject: str) -> StarterSummary:
    normalized_subject = subject.strip() or "terminal builder"
    return StarterSummary(
        title="Python CLI/TUI Starter",
        greeting=build_greeting(normalized_subject),
        commands=(
            "uv run --package python-app python-app --help",
            "uv run --package python-app python-app-tui",
            "uv run --package python-app pytest apps/python/tests",
        ),
    )


def render_summary_text(summary: StarterSummary) -> str:
    command_lines = "\n".join(f"- {command}" for command in summary.commands)
    return f"{summary.greeting}\n\nStarter commands:\n{command_lines}"
